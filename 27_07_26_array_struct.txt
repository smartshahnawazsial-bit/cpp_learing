#include<iostream>
#include<string>

using namespace std;

struct Student 
{
	int id;
	string name;
	double marks;

};

int main()
{
	Student students[3];
	
  
	for(int i = 0; i < 3; i++)
	{
		cout << "Enter student " << i+1 << endl;
		
		cout << "ID: ";
		cin >> students[i].id;
	

		cout << "Name: ";
	//	cin >> students[i].name;
		getline(cin , students[i].name);
		
		cout << "Marks: ";
		cin >> students[i].marks;	
		}
		
		cout << "\nStudent Records\n";
		
		for (int i = 0; i < 3; i++)
		{
			cout << students[i].id << " " << students[i].name << " " << students[i].marks << endl;
			   
						
		}
		return 0;
}
